{{- /* Builds the learning path from data/learning_path.yaml and the guides shown in the current language. Call it as: partialCached "learning-path.html" . site.Language.Lang (it does not depend on the page it is called from) Returns a dict: stages - the stages in order. Each has: slug, title, level, rank (1-4), blurb, topics, done, total, extra. Titles and blurbs are already in the current language. topics is a list of {key, title, pages}; pages is empty while the topic is only planned. done = topics with at least one published guide. extra = true for the "Other guides" stage. count - number of real stages (not counting "Other guides") default - slug of the stage to open first: the first one that has a published guide */ -}} {{- $data := hugo.Data.learning_path -}} {{- if not $data -}}{{- errorf "data/learning_path.yaml is missing. The Guides page is built from it." -}}{{- end -}} {{- $articles := partialCached "guides.html" . site.Language.Lang -}} {{- $levels := dict "beginner" 1 "intermediate" 2 "advanced" 3 "expert" 4 -}} {{- $known := slice -}} {{- $stages := slice -}} {{- $default := "" -}} {{- range $data.stages -}} {{- $s := . -}} {{- $topics := slice -}} {{- $done := 0 -}} {{- range $s.topics -}} {{- $known = $known | append .key -}} {{- $pages := sort (where $articles "Params.topic" .key) "Params.updated" "desc" -}} {{- if $pages -}} {{- $done = add $done 1 -}} {{- if not $default }}{{ $default = $s.slug }}{{ end -}} {{- end -}} {{- $topics = $topics | append (dict "key" .key "title" (partial "pick-lang.html" .title) "pages" $pages) -}} {{- end -}} {{- $rank := index $levels $s.level | default 1 -}} {{- $stages = $stages | append (dict "slug" $s.slug "title" (partial "pick-lang.html" $s.title) "level" $s.level "rank" $rank "blurb" (partial "pick-lang.html" $s.blurb) "topics" $topics "done" $done "total" (len $s.topics) "extra" false) -}} {{- end -}} {{- /* A guide whose topic is not on the path must not vanish: list it under "Other guides" and warn. */ -}} {{- $extras := slice -}} {{- range $articles -}} {{- $t := .Params.topic | default "" -}} {{- if not (in $known $t) -}} {{- warnf "Guide %s has topic %q, which is not in data/learning_path.yaml. It is listed under Other guides." .RelPermalink $t -}} {{- $extras = $extras | append (dict "key" $t "title" .Title "pages" (slice .)) -}} {{- end -}} {{- end -}} {{- if $extras -}} {{- $stages = $stages | append (dict "slug" "other" "title" (i18n "other_guides_title") "level" "" "rank" 0 "blurb" (i18n "other_guides_blurb") "topics" $extras "done" (len $extras) "total" (len $extras) "extra" true) -}} {{- end -}} {{- if and (not $default) $stages -}} {{- with index $stages 0 }}{{ $default = .slug }}{{ end -}} {{- end -}} {{- return (dict "stages" $stages "count" (len $data.stages) "default" $default) -}}
